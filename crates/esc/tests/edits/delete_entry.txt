== plate.esc
esc 1
part

width = param(60 mm)
  why: "fits the 58 mm motor spacing"

thick = param(4 mm)

density = param(1.24 g/cm3)

outline = sketch(on: plane(XY))
  base = line(~(0 mm, 0 mm), ~(60 mm, 0 mm))
  side = line(base.end, ~(60 mm, 40 mm))
  top = line(side.end, ~(0 mm, 40 mm))
  back = line(top.end, base.start)
    why: "closes the outline"
  c1 = coincident(base.start, origin)
  c2 = horizontal(base)
  c3 = vertical(side)
  c4 = horizontal(top)
  w = length(base, width)
  h = length(side, 40 mm)

plate = extrude(outline, thick)

hole_at = point(x: width / 2 - 6 mm, y: 0 mm, z: thick)

drilled = hole(plate, d: 3.2 mm, at: hole_at, along: -Z, depth: thick)

motor_mount = connector(origin: hole_at, z: +Z, x: +Y)
